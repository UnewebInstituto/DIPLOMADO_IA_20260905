```python
# Let's do decomposition or trend + seasonality analysis
import matplotlib.pyplot as plt
import seaborn as sns

# Moving average for trend (52-week rolling mean for annual seasonality)
weekly_total['Trend'] = weekly_total['Weekly_Sales'].rolling(window=52, center=True, min_periods=1).mean()

# Fill NaNs in trend with linear interpolation or backfill/ffill
weekly_total['Trend'] = weekly_total['Trend'].bfill().ffill()

# Calculate seasonal component (Ratio: Actual / Trend, or Difference: Actual - Trend)
# Let's use multiplicative or additive. Multiplicative is standard for retail.
weekly_total['Ratio'] = weekly_total['Weekly_Sales'] / weekly_total['Trend']

# Extract month/week of year to see seasonality
weekly_total['WeekOfYear'] = weekly_total.index.isocalendar().week
weekly_total['Month'] = weekly_total.index.month

# Average seasonal index by week of year
seasonal_index = weekly_total.groupby('WeekOfYear')['Ratio'].mean()

# Normalize seasonal index so its mean is 1
seasonal_index = seasonal_index / seasonal_index.mean()

print("Seasonal index sample:", seasonal_index.head())
```

```python
# Let's fit a simple trend model (linear regression on time index)
from sklearn.linear_model import LinearRegression

weekly_total['TimeIndex'] = np.arange(len(weekly_total))
X = weekly_total[['TimeIndex']]
y = weekly_total['Weekly_Sales']

model = LinearRegression()
model.fit(X, y)
trend_slope = model.coef_[0]
trend_intercept = model.intercept_

print(f"Trend slope: {trend_slope}, intercept: {trend_intercept}")

# Future 3 months = ~13 weeks (from last date 2012-10-26)
last_date = weekly_total.index.max()
future_dates = pd.date_range(start=last_date + pd.Timedelta(weeks=1), periods=13, freq='W-FRI')
future_time_index = np.arange(len(weekly_total), len(weekly_total) + 13)

future_trend = trend_intercept + trend_slope * future_time_index

# Get seasonal index for future weeks
# map week of year
future_weeks = future_dates.isocalendar().week
# Handle week 53 if present or map carefully
future_seasonal = [seasonal_index.get(w, 1.0) for w in future_weeks]

# Baseline projection (Expected)
expected_sales = future_trend * future_seasonal

# Scenarios:
# Optimist: Expected + 10% or trend + upper band
# Pessimist: Expected - 10% or trend - lower band
optimistic_sales = expected_sales * 1.08
pessimistic_sales = expected_sales * 0.92

future_df = pd.DataFrame({
    'Date': future_dates,
    'Esperado': expected_sales,
    'Optimista': optimistic_sales,
    'Pesimista': pessimistic_sales
})

print(future_df)
```

## ⚙️ 2. Supuestos Utilizados para las Proyecciones

Para proyectar las siguientes 13 semanas (noviembre 2012 a enero 2013) se establecieron los siguientes supuestos:

1. **Modelo Multiplicativo:** La proyección combina la tendencia proyectada (prolongación de la línea base histórica) multiplicada por el **índice estacional** histórico de cada semana del año.
2. **Escenario Esperado:** Sigue estrictamente la tendencia histórica ajustada por el factor estacional tradicional (capturando los picos típicos de noviembre/diciembre y la caída de enero).
3. **Escenario Optimista:** Supone un incremento del **$+8\%$** sobre el escenario esperado, reflejando un entorno macroeconómico favorable, mayor confianza del consumidor y mejor desempeño en las semanas de ofertas (Black Friday y Navidad).
4. **Escenario Pesimista:** Supone una contracción del **$-8\%$** respecto al escenario esperado, considerando posibles presiones inflacionarias, mayor desempleo o menor poder adquisitivo.
